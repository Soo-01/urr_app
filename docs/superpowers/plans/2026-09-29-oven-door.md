## Review Focus

스펙이 전제하지만 어느 태스크의 테스트도 직접 다루지 않는, 환자에게 실제로 문제가 될 입력들이다. 각 줄의 테스트는 해당 태스크에 포함해 두었다.

1. **큰 `dt` 한 번에 여러 전이** — 앱이 백그라운드에 갔다 오거나 프레임이 밀리면 `dt`가 수 초로 들어온다. 굽기가 끝나고 여유 시간까지 지난 상황이면 벨과 탐이 같은 프레임에 나와야 하고, 진행 막대가 한 프레임에 1.0을 넘어도 판정은 한 번만 나와야 한다. (Task 2)
2. **벨이 울릴 때 이미 문이 열려 있는 경우** — 꺼내기 단계가 시작되자마자 요구 열림을 넘고 있으면 그 프레임부터 유지가 시작돼야 한다. 닫았다 다시 열라고 요구하면 환자가 멈춘다. (Task 2)
3. **정규화 값이 범위를 벗어남** — 캘리브레이션 밖 각도나 `romRatio` 나눗셈으로 문 열림이 1을 넘거나 음수가 될 수 있다. 0~1로 잘라야 한다. (Task 2)
4. **캘리브레이션 가동범위가 0인 환자** — `AngleNormalizer`는 min과 max가 같으면 항상 0.5를 돌려준다. 문이 절반쯤 열린 채 멈추므로 요구 열림 55% 이상 레벨은 영원히 성공할 수 없다. 게임이 죽지 않고 진행되며 8분 상한으로 끝나야 한다. (Task 2)
5. **저항 설정값이 범위를 벗어남** — 허브가 3kg 초과나 음수를 넘기면 0~3으로 잘라야 하고, 4단계 미만이면 값과 무관하게 0이어야 한다. 모터에 잘못된 힘이 걸리면 안전 문제다. (Task 1)

import 'dart:math';

import '../game_base.dart';

/// ============================================================================
/// [E3] 오븐 문 열기 — 게임 규칙
/// 순수 Dart (Flame·Flutter import 금지) → flutter test로 기기 없이 검증한다.
/// 설계: docs/superpowers/specs/2026-09-29-oven-door-design.md
/// ============================================================================

enum FoodKind { cookie, muffin, bread, pizza, cake }

extension FoodKindLabel on FoodKind {
  String get label => switch (this) {
        FoodKind.cookie => '쿠키',
        FoodKind.muffin => '머핀',
        FoodKind.bread => '식빵',
        FoodKind.pizza => '피자',
        FoodKind.cake => '케이크',
      };
}

/// 규칙 상수. 실기기 QA에서 이 값들을 조정한다.
class OvenRules {
  OvenRules._();

  /// 이 값보다 적게 열려 있으면 문이 닫힌 것으로 본다 (굽기 시작 조건).
  static const double closedBelow = 0.15;

  /// 반복 횟수를 셀 때 쓰는 열림/닫힘 히스테리시스.
  static const double openAt = 0.18;
  static const double closeAt = 0.12;

  /// 손떨림 유예. 요구 열림 밑으로 내려가도 이 시간 안에 되돌아오면 이어진다.
  static const double graceTime = 0.3;

  static const double bakeTime = 8.0; // 굽는 시간(휴식)
  static const double resultTime = 2.0; // 결과 연출
  static const double maxSessionSeconds = 480; // 8분

  static const int insertScore = 50;
  static const int takeScore = 100;
  static const int speedBonusMax = 50;

  static const double maxResistanceKg = 3.0;
}

/// GameConfig와 치료사가 고른 저항에서 뽑아낸 수치 (스펙 6장).
class OvenParams {
  final FoodKind food;

  /// 음식이 들어갈 만큼 열어야 하는 문 열림 비율 (0~1).
  final double requiredOpen;

  /// 그만큼 열고 버텨야 하는 시간(초).
  final double holdTime;

  /// 벨이 울린 뒤 음식이 타기까지의 시간(초).
  final double takeOutWindow;

  final int foodGoal;
  final double resistanceKg;

  const OvenParams({
    required this.food,
    required this.requiredOpen,
    required this.holdTime,
    required this.takeOutWindow,
    required this.foodGoal,
    required this.resistanceKg,
  });

  factory OvenParams.fromConfig(GameConfig config, {double resistanceKg = 0}) {
    final level = config.difficultyLevel.clamp(1, 5);
    const foods = {
      1: FoodKind.cookie,
      2: FoodKind.muffin,
      3: FoodKind.bread,
      4: FoodKind.pizza,
      5: FoodKind.cake,
    };
    const required = {1: 0.35, 2: 0.45, 3: 0.55, 4: 0.70, 5: 0.85};
    const hold = {1: 1.0, 2: 1.5, 3: 2.0, 4: 2.5, 5: 3.0};
    const window = {1: 16.0, 2: 14.0, 3: 12.0, 4: 10.0, 5: 8.0};
    const goal = {1: 4, 2: 5, 3: 6, 4: 7, 5: 8};

    final allowed = config.canUseResistance ? resistanceKg : 0.0;
    return OvenParams(
      food: foods[level]!,
      requiredOpen: required[level]!,
      holdTime: hold[level]!,
      takeOutWindow: window[level]!,
      foodGoal: goal[level]!,
      resistanceKg: allowed.clamp(0.0, OvenRules.maxResistanceKg).toDouble(),
    );
  }
}

const _params = OvenParams(
    food: FoodKind.bread,
    requiredOpen: 0.5,

    foodGoal: 1,
    resistanceKg: 0);

/// door를 seconds 동안 60fps로 유지하고, 그동안 나온 이벤트 이름을 돌려준다.
List<String> _hold(OvenSession s, double door, double seconds) {
  final names = <String>[];
  for (var i = 0; i < (seconds * 60).round(); i++) {
    names.addAll(s.update(1 / 60, door).map((e) => e.name));
  }
  return names;
}

/// 넣기를 끝내고 문을 닫아 굽기까지 마친 뒤, 벨이 울린 직후 상태로 만든다.
OvenSession _atTakeOut({OvenParams params = _params}) {
  final s = OvenSession(params);
  _hold(s, 0.9, params.holdTime + 0.2); // 열고 유지 → 넣기 완료
  _hold(s, 0.0, 0.1); // 문 닫기 → 굽기 시작
  _hold(s, 0.0, OvenRules.bakeTime + 0.1); // 굽는 동안 대기 → 벨
  return s;
}

/// 화면에 알리는 이벤트. name은 AngleRecord.event에 그대로 기록한다.
class OvenEvent {
  final String name;
  const OvenEvent(this.name);
}

enum OvenPhase {
  inserting, // 문을 열고 버텨 음식을 넣는 중
  closing, // 넣기 완료 → 문이 닫히길 기다림
  baking, // 굽는 중 (팔을 내리고 휴식)
  takingOut, // 벨이 울린 뒤 꺼내는 중
  result, // 결과 연출
  finished,
}

enum FoodOutcome { done, dropped, burnt }

/// 한 판의 진행 상태. 화면은 매 프레임 update(dt, door)를 부르고 이벤트로 연출한다.
class OvenSession {
  final OvenParams params;

  OvenSession(this.params);

  OvenPhase _phase = OvenPhase.inserting;
  double _phaseTime = 0;
  double _elapsed = 0;
  double _door = 0;
  bool _doorOpen = false;
  double _progress = 0; // 유지 진행 0~1
  double _grace = 0; // 요구 열림 아래로 내려간 시간
  int _foodsDone = 0;
  int _score = 0;
  int _reps = 0;
  int _hits = 0;
  int _misses = 0;
  FoodOutcome? _lastOutcome;
  final List<FoodOutcome> _shelf = [];

  OvenPhase get phase => _phase;
  double get phaseTime => _phaseTime;
  double get holdProgress => _progress;
  double get door => _door;
  bool get doorOpen => _doorOpen;
  double get bakeRemaining =>
      _phase == OvenPhase.baking ? (OvenRules.bakeTime - _phaseTime).clamp(0.0, OvenRules.bakeTime) : 0;
  double get windowRemaining => _phase == OvenPhase.takingOut
      ? (params.takeOutWindow - _phaseTime).clamp(0.0, params.takeOutWindow)
      : 0;
  int get foodsDone => _foodsDone;
  int get score => _score;
  int get reps => _reps;
  int get hits => _hits;
  int get misses => _misses;
  double get elapsed => _elapsed;
  FoodOutcome? get lastOutcome => _lastOutcome;
  List<FoodOutcome> get shelf => List.unmodifiable(_shelf);
  bool get isFinished => _phase == OvenPhase.finished;
  double get accuracy => (_hits + _misses) == 0 ? 0 : _hits / (_hits + _misses);

  int get maxPossibleScore =>
      params.foodGoal *
      (OvenRules.insertScore + OvenRules.takeScore + OvenRules.speedBonusMax);

  /// door = 문 열림 0~1. 이번 프레임에 생긴 이벤트를 돌려준다.
  List<OvenEvent> update(double dt, double door) {
    final events = <OvenEvent>[];
    if (_phase == OvenPhase.finished) return events;

    _door = door.isFinite ? door.clamp(0.0, 1.0) : 0.0;
    _elapsed += dt;
    _phaseTime += dt;
    _updateDoorState(events);

    switch (_phase) {
      case OvenPhase.inserting:
        _hold(dt, events, onDone: () {
          _hits++;
          _score += OvenRules.insertScore;
          events.add(const OvenEvent('insert_done'));
          _enter(OvenPhase.closing);
        });
      case OvenPhase.closing:
        if (_door < OvenRules.closedBelow) _enter(OvenPhase.baking);
      case OvenPhase.baking:
        if (_phaseTime >= OvenRules.bakeTime) {
          // 큰 dt로 굽는 시간을 한참 넘겨 들어올 수 있다. 넘긴 만큼을 꺼내기
          // 단계의 경과 시간으로 넘겨야 같은 프레임에 탐 판정까지 난다.
          final overshoot = _phaseTime - OvenRules.bakeTime;
          events.add(const OvenEvent('bell'));
          _enter(OvenPhase.takingOut);
          _phaseTime = overshoot;
          // 벨이 울린 프레임에도 이미 열려 있으면 바로 유지를 시작한다.
          _takeOut(0, events);
        }
      case OvenPhase.takingOut:
        _takeOut(dt, events);
      case OvenPhase.result:
        if (_phaseTime >= OvenRules.resultTime) _nextFood();
      case OvenPhase.finished:
        break;
    }

    if (_phase != OvenPhase.finished && _elapsed >= OvenRules.maxSessionSeconds) {
      _enter(OvenPhase.finished);
    }
    return events;
  }

  void _enter(OvenPhase phase) {
    _phase = phase;
    _phaseTime = 0;
    _progress = 0;
    _grace = 0;
  }

  void _updateDoorState(List<OvenEvent> events) {
    if (!_doorOpen && _door >= OvenRules.openAt) {
      _doorOpen = true;
      events.add(const OvenEvent('door_open'));
    } else if (_doorOpen && _door < OvenRules.closeAt) {
      _doorOpen = false;
      _reps++;
      events.add(const OvenEvent('door_close'));
    }
  }

  /// 유지 진행. 요구 열림 아래로 내려가면 유예 시간을 재고, 넘기면 음식을 떨어뜨린다.
  void _hold(double dt, List<OvenEvent> events, {required void Function() onDone}) {
    if (_door >= params.requiredOpen) {
      _grace = 0;
      _progress += dt / params.holdTime;
      if (_progress >= 1.0) {
        _progress = 1.0;
        onDone();
      }
      return;
    }
    if (_progress <= 0) return; // 아직 시작하지 않았다
    _grace += dt;
    if (_grace > OvenRules.graceTime) {
      _misses++;
      events.add(const OvenEvent('drop'));
      _finishFood(FoodOutcome.dropped);
    }
  }

  void _takeOut(double dt, List<OvenEvent> events) {
    if (_phase != OvenPhase.takingOut) return;
    _hold(dt, events, onDone: () {
      final remaining =
          (params.takeOutWindow - _phaseTime).clamp(0.0, params.takeOutWindow);
      _hits++;
      _score += OvenRules.takeScore +
          (OvenRules.speedBonusMax * remaining / params.takeOutWindow).round();
      events.add(const OvenEvent('take_done'));
      _shelf.add(FoodOutcome.done);
      _finishFood(FoodOutcome.done);
    });
    if (_phase == OvenPhase.takingOut && _phaseTime >= params.takeOutWindow) {
      _misses++;
      events.add(const OvenEvent('burnt'));
      _finishFood(FoodOutcome.burnt);
    }
  }

  void _finishFood(FoodOutcome outcome) {
    _lastOutcome = outcome;
    _foodsDone++;
    _enter(OvenPhase.result);
  }

  void _nextFood() {
    if (_foodsDone >= params.foodGoal) {
      _enter(OvenPhase.finished);
      return;
    }
    _enter(OvenPhase.inserting);
  }
}

class OvenDoorFlameGame extends FlameGame {

  bool _started = false;
  bool paused = false;
  bool _ended = false;
  double _recordClock = 0;
  final List<AngleRecord> _history = [];

  });

  double get doorOpen => (_p / config.romRatio).clamp(0.0, 1.0);

  @override
  Future<void> onLoad() async {
    session = OvenSession(OvenParams.fromConfig(config, resistanceKg: resistanceKg));
    _sub = rawAngleStream?.listen((a) {
      _raw = a;
      _p = config.normalizer.normalize(a);
    });

    _p = v.clamp(0.0, 1.0);
    _raw = config.normalizer.denormalize(_p);

  @override

    super.update(dt);
    if (!isLoaded || _ended || paused) return;
    for (final p in _popups) {
      p.life += dt;
    }
    _popups.removeWhere((p) => p.life > 1.4);

    if (!_started) {
      _countdown -= dt;
      if (_countdown <= 0) _started = true;
      return;
    }
    final events = session.update(dt, doorOpen);
    for (final e in events) {
      _onEvent(e);
    }
    _record(dt, events);
    if (session.isFinished) endGame();

      case 'insert_done':
        _popups.add(_Popup('들어갔어요!', _good));

      case 'burnt':
        _popups.add(_Popup('탔어요', _warn));
    }

    _recordClock += dt;

      timestampMs: (session.elapsed * 1000).round(),
      rawAngle: _raw,
      normalizedPosition: _p,
      targetPosition: session.params.requiredOpen,
      event: events.map((e) => e.name).join('|'),
    ));

      calibrationMax: config.normalizer.maxAngle,
      hits: session.hits,
      misses: session.misses,

    ));

      canvas.drawRRect(banner, _outline(4));
      _text(canvas, '팔꿈치를 90°로 굽힌 채 유지해 주세요', const Offset(300, 119), 22, _ink,
          center: true, bold: true);
    }
    if (paused) _drawOverlay(canvas, '일시정지', null);
    if (!_started) {
      _drawOverlay(canvas, '${_countdown.ceil().clamp(1, 3)}', '문 손잡이를 잡고 준비하세요!');
    }

  void _drawBackground(Canvas c) {

  void _drawShelf(Canvas c) {

  void _drawOvenBody(Canvas c) {
    c.drawRRect(RRect.fromRectAndRadius(_ovenRect, const Radius.circular(20)),

  void _drawFood(Canvas c) {
    final inner = _ovenRect.deflate(40);
    final phase = session.phase;
    if (phase == OvenPhase.inserting) {
      // 트레이가 진행도만큼 안으로 들어간다
      final x = inner.right - (inner.width - 60) * session.holdProgress;
      _drawFoodShape(c, Offset(x, inner.center.dy), 1.0, false);
    } else if (phase == OvenPhase.closing || phase == OvenPhase.baking) {
      _drawFoodShape(c, inner.center, 1.0, false);
    } else if (phase == OvenPhase.takingOut) {
      final x = inner.center.dx + (inner.width * 0.5 + 60) * session.holdProgress;
      _drawFoodShape(c, Offset(x, inner.center.dy), 1.0, true);
    } else if (phase == OvenPhase.result) {
      final outcome = session.lastOutcome;
      if (outcome == FoodOutcome.dropped) {
        final t = (session.phaseTime / OvenRules.resultTime).clamp(0.0, 1.0);
        _drawFoodShape(c, Offset(inner.right + 80 + 120 * t, 780), 1.0, false);
      } else if (outcome == FoodOutcome.burnt) {
        _drawFoodShape(c, inner.center, 1.0, false, burnt: true);
      }
    }
  }

  void _drawFoodShape(Canvas c, Offset at, double scale, bool hot, {bool burnt = false}) {
    final size = 90.0 * scale * (0.6 + 0.5 * session.params.requiredOpen);
    final color = burnt ? const Color(0xFF212121) : const Color(0xFFD7A15C);
    final rect = Rect.fromCenter(center: at, width: size, height: size * 0.7);
    c.drawRRect(
        RRect.fromRectAndRadius(rect, Radius.circular(size * 0.2)), Paint()..color = color);
    c.drawRRect(RRect.fromRectAndRadius(rect, Radius.circular(size * 0.2)), _outline(5));
    if (hot && !burnt) {
      for (var i = 0; i < 3; i++) {
        final x = at.dx - size * 0.25 + i * size * 0.25;
        c.drawLine(Offset(x, rect.top - 10), Offset(x + 8, rect.top - 36),
            Paint()
              ..color = Colors.white70
              ..strokeWidth = 4
              ..strokeCap = StrokeCap.round);
      }
    }

  void _drawDoor(Canvas c) {
    final angle = doorOpen * pi / 2.2; // 0 = 닫힘, 최대 약 82°
    c.save();
    c.translate(_hinge.dx, _hinge.dy);
    c.rotate(angle * 0.45); // 정면 기준 기울기
    final door = Rect.fromLTWH(0, -_doorLen / 2, _doorLen * (1 - 0.35 * doorOpen), _doorLen);
    c.drawRRect(RRect.fromRectAndRadius(door, const Radius.circular(16)),

    c.drawRRect(RRect.fromRectAndRadius(door, const Radius.circular(16)), _outline(7));
    // 유리창

  void _drawHoldBar(Canvas c) {

  late OvenDoorFlameGame _game;
  late GameMotorController _motor;
  double _simValue = 0.0;
  bool _isSim = false;

  @override
  void initState() {
    super.initState();
    _isSim = !widget.bluetoothService.isConnected();
    _motor = GameMotorController(bt: widget.bluetoothService);
    _game = OvenDoorFlameGame(
      rawAngleStream: _isSim ? null : _motor.angleStream,
      config: widget.config,
      resistanceKg: widget.resistanceKg,
      showElbowGuide: !_isSim,
      onGameEnd: (r) {
        _motor.safeStop();
        _motor.dispose();
        if (mounted) {
          Navigator.pushReplacement(
              context, MaterialPageRoute(builder: (_) => GameResultScreen(result: r)));
        }
      },
    );
    if (!_isSim) {
      _motor.selectJoint(widget.config.bodyPart);
      _motor.startWatchdog();
      final resistance = widget.config.canUseResistance
          ? widget.resistanceKg.clamp(0.0, OvenRules.maxResistanceKg)
          : 0.0;

4. 넣은 뒤 문을 닫으면 굽기 8초가 시작되고, 끝나면 "딩동" 문구와 남은 시간이 보인다.
5. 여유 시간 안에 다시 열어 버티면 음식이 나와 오른쪽 선반에 쌓이고, 시간을 넘기면 까맣게 탄다.
6. Brunnstrom 3단계를 고르면 저항 선택이 비활성화된다.
